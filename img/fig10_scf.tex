\tikzstyle{startstop} = [rectangle, rounded corners, minimum width=3cm, minimum height=1cm,text centered, draw=black, fill=black!90, text=white]
\tikzstyle{process} = [rectangle, minimum width=3cm, minimum height=1cm, text centered, draw=black, fill=black!10]
\tikzstyle{decision} = [diamond, minimum width=3cm, minimum height=1cm, text centered, draw=black, fill=black!80, text=white]
\tikzstyle{arrow} = [thick,->,>=stealth]

\begin{tikzpicture}[node distance=2cm]

\node (start) [startstop] {Choose initial $\rho_{\text{in}}$};
\node (hamiltonian) [process, below of=start, align=center]
    {Construct the fixed Hamiltonian\\$\mat{H}[\rho_{\text{in}}]$};
\node (eigensolver) [process, below of=hamiltonian, align=center]
    {Inner Arnoldi/Lanczos solve\\until $\|\mat{H}\vec{u}_{i}-\theta_i\vec{u}_{i}\|<\tau_{\text{eig}}$};
\node (density) [process, below of=eigensolver, align=center]
    {Construct $\rho_{\text{out}}$ and\\calculate $E_{\text{tot}}$};
\node (converged) [decision, below of=density, yshift=-1cm, align=center]
    {SCF\\converged?};
\node (mix) [process, right of=converged, xshift=4cm, align=center]
    {Mix $\rho_{\text{in}}$ and\\$\rho_{\text{out}}$};
\node (stop) [startstop, below of=converged, yshift=-1cm] {Done};

\draw [arrow] (start) -- (hamiltonian);
\draw [arrow] (hamiltonian) -- (eigensolver);
\draw [arrow] (eigensolver) -- (density);
\draw [arrow] (density) -- (converged);
\draw [arrow] (converged) -- node[anchor=east] {Yes} (stop);
\draw [arrow] (converged) -- node[anchor=north] {No} (mix);
\draw [arrow] (mix) |- (hamiltonian);

\end{tikzpicture}
